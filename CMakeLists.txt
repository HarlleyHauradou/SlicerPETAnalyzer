cmake_minimum_required(VERSION 3.16.3)
project(SlicerPETAnalyzer)
set(EXTENSION_HOMEPAGE "https://github.com/HarlleyHauradou/SlicerPETAnalyzer")
set(EXTENSION_CATEGORY "Quantification")
set(EXTENSION_CONTRIBUTORS "Harlley Hauradou")
set(EXTENSION_DESCRIPTION "Research image-quality analysis of IEC/NEMA hot-sphere phantom PET and quantitative SPECT volumes.")
set(EXTENSION_ICONURL "${EXTENSION_HOMEPAGE}/raw/main/PETAnalyzer/Resources/PETAnalyzer.png")
set(EXTENSION_SCREENSHOTURLS "")
set(EXTENSION_DEPENDS "NA")
find_package(Slicer REQUIRED)
include(${Slicer_USE_FILE})
add_subdirectory(PETAnalyzer)
include(${Slicer_EXTENSION_GENERATE_CONFIG})
include(${Slicer_EXTENSION_CPACK})
